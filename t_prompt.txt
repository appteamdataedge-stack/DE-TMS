showInfoBar = false;
private createRoutes = [
  '/mm-deal/create', '/portfolio/create', '/partners/create',
  '/counterparties/create', '/instruments/create',
  '/gl-master/create', '/products/create', '/ssi/create'
];
ngOnInit() {
  this.checkRoute(this.router.url);
  this.router.events.pipe(
    filter(e => e instanceof NavigationEnd)
  ).subscribe((e: NavigationEnd) => this.checkRoute(e.urlAfterRedirects));
}
checkRoute(url: string) {
  this.showInfoBar = this.createRoutes.some(r => url.startsWith(r));
}

In shell.component.html use *ngIf:
